abstract class BaseDelegate {
    +callback: Callback
    +check_asset() Asset
    +check_orders() List<Order]
    +check_positions() List[Position]
    +order_market_open(code, price, volume, remark)
    +order_market_close(code, price, volume, remark)
    +order_limit_open(code, price, volume, remark)
    +order_limit_close(code, price, volume, remark)
    +order_cancel_all(strategy_name)
    +is_position_holding(position) bool
    +get_holding_position_count(positions) int
    +shutdown()
}

class XtDelegate {
    -ding_messager: DingMessager
    -stock_names: StockNames
    -subscriber: XtSubscriber
    -xt_trader: XtQuantTrader
    -account: StockAccount
    +__init__(account_id, client_path, callback)
    +connect(callback) XtQuantTrader
    +reconnect()
    +keep_connected()
}

class GmDelegate {
    -gm_trader: GmTrader
    -account: Account
    +__init__(client_token, account_id)
    +connect()
    +check_asset() Asset
    +check_positions() List[Position]
}

class XtCallback {
    -delegate: XtDelegate
    +on_disconnected()
    +on_stock_asset(asset)
    +on_stock_order(order)
    +on_stock_trade(trade)
    +on_stock_position(position)
}

BaseDelegate <|-- XtDelegate
BaseDelegate <|-- GmDelegate
XtDelegate --> XtSubscriber : uses
XtDelegate --> DingMessager : uses
XtDelegate --> StockNames : uses
XtDelegate --> XtCallback : uses

@enduml
