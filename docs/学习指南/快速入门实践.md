新建 `selector/select_tech.py`：

```python
import akshare as ak
from tools.utils_basic import debug

def get_tech_codes():
    """基于技术指标选股"""
    try:
        # 获取所有A股列表
        stock_list = ak.stock_info_a_code_name()

        selected_codes = []

        for _, row in stock_list.iterrows():
            code = row['code']
            name = row['name']

            # 过滤ST股票
            if 'ST' in name or '退' in name:
                continue

            # 获取K线数据
            df = ak.stock_zh_a_hist(symbol=code, period="daily",
                                   start_date="20240101", end_date="20241201")

            if len(df) < 20:
                continue

            # 计算均线
            df['MA5'] = df['收盘'].rolling(5).mean()
            df['MA20'] = df['收盘'].rolling(20).mean()

            # 最新数据
            latest = df.iloc[-1]

            # 均线多头排列条件
            if (latest['MA5'] > latest['MA20'] and
                df.iloc[-2]['MA5'] > df.iloc[-2]['MA20']):

                selected_codes.append(code)
                debug(f"技术选股: {code} {name}")

        return selected_codes

    except Exception as e:
        debug(f"技术选股错误: {e}")
        return []
```

### 5.2 简单回测逻辑

```python
# 新建 tools/simple_backtest.py
import pandas as pd
import numpy as np

def simple_backtest(data, buy_signals, sell_signals):
    """简单回测函数"""
    positions = {}
    cash = 100000  # 初始资金
    results = []

    for date, signals in buy_signals.items():
        for code, price in signals.items():
            if code in data and len(data[code]) > 0:
                if cash > price * 100:  # 买入100股
                    positions[code] = {
                        'shares': 100,
                        'cost': price,
                        'date': date
                    }
                    cash -= price * 100

    # 卖出逻辑
    for date, signals in sell_signals.items():
        for code, price in signals.items():
            if code in positions:
                shares = positions[code]['shares']
                cost = positions[code]['cost']
                profit = (price - cost) * shares

                cash += price * shares
                results.append({
                    'code': code,
                    'buy_date': positions[code]['date'],
                    'sell_date': date,
                    'profit': profit,
                    'profit_rate': profit / (cost * shares)
                })

                del positions[code]

    return results
```

### 6.2 动态止损策略

```python
# 在 seller.py 中添加动态止损
def calculate_dynamic_stop_loss(current_price, highest_price, days_held):
    """动态止损计算"""
    base_stop_loss = 0.04  # 基础止损4%

    # 持仓超过3天，止损线每天上移0.1%
    if days_held > 3:
        stop_loss = base_stop_loss + (days_held - 3) * 0.001
    else:
        stop_loss = base_stop_loss

    # 从最高点回撤超过3%也卖出
    fall_stop = (highest_price - current_price) / highest_price

    return max(stop_loss, fall_stop)
```

    # 读取持仓信息
    with open('positions.json', 'r') as f:
        positions = json.load(f)

    # 计算当日收益
